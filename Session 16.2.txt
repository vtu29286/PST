import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int T = sc.nextInt();

        while (T-- > 0) {
            String s = sc.next();

            int n = s.length();
            int[] left = new int[26];
            int[] right = new int[26];
            int mid = n / 2;
            for (int i = 0; i < mid; i++) {
                left[s.charAt(i) - 'a']++;
            }
            for (int i = (n + 1) / 2; i < n; i++) {
                right[s.charAt(i) - 'a']++;
            }
            boolean isLapindrome = true;

            for (int i = 0; i < 26; i++) {
                if (left[i] != right[i]) {
                    isLapindrome = false;
                    break;
                }
            }

            System.out.println(isLapindrome ? "YES" : "NO");
        }

        sc.close();
    }
}
