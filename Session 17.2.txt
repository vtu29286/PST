import java.util.*;

class Solution {
    public boolean containsDuplicate(int[] nums) {
        HashSet<Integer> set = new HashSet<>();

        for (int num : nums) {
            // If the number is already present, it's a duplicate
            if (set.contains(num)) {
                return true;
            }

            set.add(num);
        }

        return false;
    }
}
